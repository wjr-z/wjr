cmake_minimum_required(VERSION 3.20)
project(wjr_required_component LANGUAGES CXX)
set(WJR_REQUEST_COMPONENT wjr_unavailable CACHE STRING "Component to request.")
if(WJR_OPTIONAL_COMPONENT)
    find_package(wjr CONFIG REQUIRED OPTIONAL_COMPONENTS ${WJR_REQUEST_COMPONENT})
    if(wjr_${WJR_REQUEST_COMPONENT}_FOUND)
        message(FATAL_ERROR "The unavailable component was reported as found.")
    endif()
else()
    find_package(wjr CONFIG REQUIRED COMPONENTS ${WJR_REQUEST_COMPONENT})
endif()
