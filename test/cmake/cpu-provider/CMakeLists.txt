cmake_minimum_required(VERSION 3.20)
project(wjr_cpu_provider_test LANGUAGES CXX)
include(CMakePackageConfigHelpers)
include(FetchContent)

if(NOT MODE MATCHES "^(installed|preexisting|alias|disabled|incompatible|source|never)$")
    message(FATAL_ERROR "Unknown CPU provider test mode: ${MODE}")
endif()

unset(WJR_TEST_CPU_PACKAGE_USED CACHE)
unset(WJR_TEST_CPU_SOURCE_USED CACHE)
set(_PACKAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/fake-package")
file(MAKE_DIRECTORY "${_PACKAGE_DIR}")
file(WRITE "${_PACKAGE_DIR}/CpuFeaturesConfig.cmake"
    "add_library(CpuFeatures::cpu_features INTERFACE IMPORTED)\n"
    "set(WJR_TEST_CPU_PACKAGE_USED TRUE CACHE INTERNAL \"test marker\")\n")
set(_VERSION 0.10.1)
if(MODE STREQUAL "incompatible")
    set(_VERSION 0.10.2)
endif()
write_basic_package_version_file("${_PACKAGE_DIR}/CpuFeaturesConfigVersion.cmake"
    VERSION ${_VERSION} COMPATIBILITY SameMajorVersion)
set(CpuFeatures_DIR "${_PACKAGE_DIR}" CACHE PATH "Test CPU package" FORCE)

# A parent declaration supplies the fallback without downloading dependencies.
set(_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/fake-source")
file(MAKE_DIRECTORY "${_SOURCE_DIR}")
file(WRITE "${_SOURCE_DIR}/CMakeLists.txt"
    "add_library(cpu_features INTERFACE)\n"
    "add_library(CpuFeatures::cpu_features ALIAS cpu_features)\n"
    "set(WJR_TEST_CPU_SOURCE_USED TRUE CACHE INTERNAL \"test marker\")\n")
set(FETCHCONTENT_TRY_FIND_PACKAGE_MODE NEVER)
FetchContent_Declare(cpu_features SOURCE_DIR "${_SOURCE_DIR}")
unset(FETCHCONTENT_TRY_FIND_PACKAGE_MODE)

if(MODE STREQUAL "preexisting")
    add_library(CpuFeatures::cpu_features INTERFACE IMPORTED)
elseif(MODE STREQUAL "alias")
    add_library(parent_cpu_features INTERFACE)
    add_library(CpuFeatures::cpu_features ALIAS parent_cpu_features)
endif()
if(MODE MATCHES "^(preexisting|alias|disabled)$")
    set(CMAKE_DISABLE_FIND_PACKAGE_CpuFeatures TRUE)
endif()
if(MODE MATCHES "^(preexisting|alias|source)$")
    set(FETCHCONTENT_SOURCE_DIR_CPU_FEATURES "${_SOURCE_DIR}" CACHE PATH "Explicit source" FORCE)
endif()
if(MODE STREQUAL "never")
    set(FETCHCONTENT_TRY_FIND_PACKAGE_MODE NEVER)
endif()

# Isolate CPU selection from unrelated vendored dependencies and compilation.
add_library(fmt::fmt INTERFACE IMPORTED GLOBAL)
set(WJR_BUILD_TESTS OFF)
set(WJR_BUILD_BENCHMARKS OFF)
set(WJR_ENABLE_INSTALL OFF)
set(WJR_ENABLE_ASSEMBLY OFF)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../.." wjr)

if(NOT TARGET CpuFeatures::cpu_features)
    message(FATAL_ERROR "CPU provider target must remain visible to its parent")
endif()
if(MODE STREQUAL "installed")
    get_target_property(_GLOBAL CpuFeatures::cpu_features IMPORTED_GLOBAL)
    if(NOT WJR_TEST_CPU_PACKAGE_USED OR WJR_TEST_CPU_SOURCE_USED OR NOT _GLOBAL)
        message(FATAL_ERROR "Installed CPU package was not reused with global visibility")
    endif()
elseif(MODE MATCHES "^(preexisting|alias)$")
    if(WJR_TEST_CPU_PACKAGE_USED OR WJR_TEST_CPU_SOURCE_USED)
        message(FATAL_ERROR "Existing parent target must bypass provider selection")
    endif()
    if(MODE STREQUAL "preexisting")
        get_target_property(_GLOBAL CpuFeatures::cpu_features IMPORTED_GLOBAL)
        if(_GLOBAL)
            message(FATAL_ERROR "WJR changed the parent imported target's scope")
        endif()
    else()
        get_target_property(_ALIASED CpuFeatures::cpu_features ALIASED_TARGET)
        if(NOT _ALIASED STREQUAL "parent_cpu_features")
            message(FATAL_ERROR "WJR replaced the parent alias")
        endif()
    endif()
elseif(WJR_TEST_CPU_PACKAGE_USED OR NOT WJR_TEST_CPU_SOURCE_USED)
    message(FATAL_ERROR "Disabled/incompatible package or forced source did not use fallback")
endif()
