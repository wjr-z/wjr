cmake_minimum_required(VERSION 3.20)
project(wjr_provider_scope LANGUAGES CXX)

# Start from the requested state even when CTest reconfigures this build directory.
unset(BUILD_TESTING)
unset(BUILD_TESTING CACHE)
set(EXPECT_NORMAL "")
set(EXPECT_CACHE "")
if(TESTING_STATE MATCHES "cache-(on|off)$")
   string(TOUPPER "${CMAKE_MATCH_1}" EXPECT_CACHE)
   set(BUILD_TESTING "${EXPECT_CACHE}" CACHE BOOL "Parent test setting")
endif()
if(TESTING_STATE MATCHES "^normal-(on|off)")
   string(TOUPPER "${CMAKE_MATCH_1}" EXPECT_NORMAL)
   set(BUILD_TESTING "${EXPECT_NORMAL}")
endif()
if(NOT TESTING_STATE MATCHES "^(undefined|cache-(on|off)|normal-(on|off)(-cache-(on|off))?)$")
   message(FATAL_ERROR "Unknown TESTING_STATE: ${TESTING_STATE}")
endif()

set(EXPECT_TESTS OFF)
if(BUILD_TESTING)
   set(EXPECT_TESTS ON)
endif()
enable_testing()
if(BUILD_TESTING)
   add_test(NAME parent_before COMMAND ${CMAKE_COMMAND} -E true)
endif()

# Exercise the real pinned provider, reusing its already populated source tree.
set(FETCHCONTENT_TRY_FIND_PACKAGE_MODE NEVER)
set(WJR_BUILD_TESTS OFF)
set(WJR_BUILD_BENCHMARKS OFF)
set(WJR_ENABLE_INSTALL OFF)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../.." wjr)
FetchContent_GetProperties(cpu_features)

get_property(HAS_CACHE CACHE BUILD_TESTING PROPERTY TYPE SET)
if(EXPECT_CACHE STREQUAL "")
   if(HAS_CACHE)
      message(FATAL_ERROR "WJR created a parent BUILD_TESTING cache entry.")
   endif()
else()
   if(NOT HAS_CACHE OR NOT "$CACHE{BUILD_TESTING}" STREQUAL EXPECT_CACHE)
      message(FATAL_ERROR "WJR changed the parent BUILD_TESTING cache value.")
   endif()
   get_property(CACHE_TYPE CACHE BUILD_TESTING PROPERTY TYPE)
   get_property(CACHE_HELP CACHE BUILD_TESTING PROPERTY HELPSTRING)
   if(NOT CACHE_TYPE STREQUAL "BOOL" OR NOT CACHE_HELP STREQUAL "Parent test setting")
      message(FATAL_ERROR "WJR changed the parent BUILD_TESTING cache metadata.")
   endif()
endif()

# Remove the cache temporarily to distinguish a normal binding from a cache lookup.
unset(BUILD_TESTING CACHE)
if(EXPECT_NORMAL STREQUAL "")
   if(DEFINED BUILD_TESTING)
      message(FATAL_ERROR "WJR created a parent BUILD_TESTING normal variable.")
   endif()
elseif(NOT DEFINED BUILD_TESTING OR NOT BUILD_TESTING STREQUAL EXPECT_NORMAL)
   message(FATAL_ERROR "WJR changed the parent BUILD_TESTING normal variable.")
endif()
if(NOT EXPECT_CACHE STREQUAL "")
   set(BUILD_TESTING "${EXPECT_CACHE}" CACHE BOOL "Parent test setting")
   # CMake 3.20's cache assignment can remove a normal binding (CMP0126 OLD).
   if(NOT EXPECT_NORMAL STREQUAL "")
      set(BUILD_TESTING "${EXPECT_NORMAL}")
   endif()
endif()

if(NOT TARGET CpuFeatures::cpu_features OR NOT TARGET wjr::arch-cpuinfo)
   message(FATAL_ERROR "The embedded project did not configure the CPU provider.")
endif()
get_directory_property(PROVIDER_TESTING DIRECTORY "${cpu_features_SOURCE_DIR}"
   DEFINITION BUILD_TESTING)
get_property(PROVIDER_SUBDIRS DIRECTORY "${cpu_features_SOURCE_DIR}" PROPERTY SUBDIRECTORIES)
get_property(PROVIDER_TESTS DIRECTORY "${cpu_features_SOURCE_DIR}" PROPERTY TESTS)
if(PROVIDER_TESTING OR PROVIDER_TESTS OR
   "${cpu_features_SOURCE_DIR}/test" IN_LIST PROVIDER_SUBDIRS OR TARGET cpuinfo_x86_test)
   message(FATAL_ERROR "The embedded CPU provider enabled its own tests.")
endif()

if(BUILD_TESTING)
   add_test(NAME parent_after COMMAND ${CMAKE_COMMAND} -E true)
endif()
add_subdirectory(sibling)
get_property(PARENT_TESTS DIRECTORY PROPERTY TESTS)
get_property(SIBLING_TESTS DIRECTORY sibling PROPERTY TESTS)
if(EXPECT_TESTS)
   if(NOT "parent_before" IN_LIST PARENT_TESTS OR NOT "parent_after" IN_LIST PARENT_TESTS OR
      NOT "sibling" IN_LIST SIBLING_TESTS)
      message(FATAL_ERROR "WJR prevented parent or sibling test registration.")
   endif()
elseif(PARENT_TESTS OR SIBLING_TESTS)
   message(FATAL_ERROR "Tests were registered despite the parent's disabled test setting.")
endif()
