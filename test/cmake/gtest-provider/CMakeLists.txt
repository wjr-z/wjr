cmake_minimum_required(VERSION 3.20)
project(wjr_gtest_provider_test LANGUAGES NONE)
include(CMakePackageConfigHelpers)
include(FetchContent)
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/wjrGTest.cmake")

if(NOT MODE MATCHES "^(installed|preexisting|alias|bare|bare-alias|disabled|incompatible|broken|source|never|local|asan|coverage|debug|multi)$")
    message(FATAL_ERROR "Unknown GoogleTest provider mode: ${MODE}")
endif()
if(MODE MATCHES "^(installed|broken|asan)$")
    enable_language(CXX)
endif()
set(CMAKE_BUILD_TYPE Release)
set(CMAKE_FIND_PACKAGE_TARGETS_GLOBAL TRUE)
unset(WJR_TEST_GTEST_PACKAGE_USED CACHE)
unset(WJR_TEST_GTEST_SOURCE_USED CACHE)
set(_PACKAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/fake-package")
file(MAKE_DIRECTORY "${_PACKAGE_DIR}/include/gtest")
file(WRITE "${_PACKAGE_DIR}/include/gtest/gtest.h" [=[
#include <string>
namespace testing {
struct Message {
    template <typename T> Message& operator<<(const T&) { return *this; }
#ifdef WJR_TEST_GTEST_BROKEN
    std::string GetString() const;
#else
    std::string GetString() const { return "wjr"; }
#endif
};
}
#define TEST(Suite, Name) int main()
#define EXPECT_EQ(a, b) (void)((a) == (b))
]=])
file(WRITE "${_PACKAGE_DIR}/GTestConfig.cmake" [=[
if(TARGET GTest::gtest AND TARGET GTest::gtest_main AND
   TARGET GTest::gmock AND TARGET GTest::gmock_main)
    return()
endif()
add_library(wjr_fake_gtest_dependency INTERFACE IMPORTED)
set_target_properties(wjr_fake_gtest_dependency PROPERTIES
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_LIST_DIR}/include")
foreach(_target GTest::gtest GTest::gtest_main GTest::gmock GTest::gmock_main)
    add_library(${_target} INTERFACE IMPORTED)
    set_target_properties(${_target} PROPERTIES
        INTERFACE_LINK_LIBRARIES wjr_fake_gtest_dependency)
endforeach()
if(MODE STREQUAL "broken")
    set_property(TARGET GTest::gtest_main PROPERTY
        INTERFACE_COMPILE_DEFINITIONS WJR_TEST_GTEST_BROKEN)
endif()
set(WJR_TEST_GTEST_PACKAGE_USED TRUE CACHE INTERNAL "test marker")
]=])
set(_VERSION 1.17.0)
if(MODE STREQUAL "incompatible")
    set(_VERSION 1.16.0)
endif()
write_basic_package_version_file("${_PACKAGE_DIR}/GTestConfigVersion.cmake"
    VERSION ${_VERSION} COMPATIBILITY SameMajorVersion)
set(GTest_DIR "${_PACKAGE_DIR}" CACHE PATH "Test GoogleTest package" FORCE)

# A parent declaration supplies the fallback without network access.
set(_SOURCE_DIR "${CMAKE_CURRENT_BINARY_DIR}/fake-source")
file(MAKE_DIRECTORY "${_SOURCE_DIR}")
file(WRITE "${_SOURCE_DIR}/empty.cpp" "int wjr_gtest_fixture;\n")
file(WRITE "${_SOURCE_DIR}/CMakeLists.txt" [=[
if(WJR_ENABLE_ASAN)
    add_library(gtest STATIC empty.cpp)
    add_library(gtest_main STATIC empty.cpp)
else()
    add_library(gtest INTERFACE)
    add_library(gtest_main INTERFACE)
endif()
add_library(GTest::gtest ALIAS gtest)
add_library(GTest::gtest_main ALIAS gtest_main)
set(WJR_TEST_GTEST_SOURCE_USED TRUE CACHE INTERNAL "test marker")
]=])
set(FETCHCONTENT_TRY_FIND_PACKAGE_MODE NEVER)
FetchContent_Declare(googletest SOURCE_DIR "${_SOURCE_DIR}")
unset(FETCHCONTENT_TRY_FIND_PACKAGE_MODE)

set(gtest_force_shared_crt sentinel CACHE STRING "Parent setting" FORCE)
set(BUILD_GMOCK sentinel CACHE STRING "Parent setting" FORCE)
set(BUILD_GTEST sentinel CACHE STRING "Parent setting" FORCE)
if(MODE STREQUAL "preexisting")
    add_library(GTest::gtest_main INTERFACE IMPORTED)
elseif(MODE STREQUAL "alias")
    add_library(parent_gtest_main INTERFACE)
    add_library(GTest::gtest_main ALIAS parent_gtest_main)
elseif(MODE STREQUAL "bare")
    add_library(gtest_main INTERFACE)
elseif(MODE STREQUAL "bare-alias")
    add_library(parent_gtest_main INTERFACE)
    add_library(gtest_main ALIAS parent_gtest_main)
elseif(MODE STREQUAL "disabled")
    set(CMAKE_DISABLE_FIND_PACKAGE_GTest TRUE)
elseif(MODE STREQUAL "source")
    set(FETCHCONTENT_SOURCE_DIR_GOOGLETEST "${_SOURCE_DIR}" CACHE PATH "Explicit source" FORCE)
elseif(MODE STREQUAL "never")
    set(FETCHCONTENT_TRY_FIND_PACKAGE_MODE NEVER)
elseif(MODE STREQUAL "asan")
    set(WJR_ENABLE_ASAN TRUE)
elseif(MODE STREQUAL "coverage")
    set(WJR_ENABLE_COVERAGE TRUE)
elseif(MODE STREQUAL "debug")
    set(CMAKE_BUILD_TYPE Debug)
elseif(MODE STREQUAL "multi")
    set(CMAKE_CONFIGURATION_TYPES "Debug;Release")
endif()
if(MODE STREQUAL "broken")
    # A compile-only toolchain default must not bypass the provider's link check.
    set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
endif()
if(MODE MATCHES "^(preexisting|alias|bare|bare-alias)$")
    set(WJR_ENABLE_ASAN TRUE)
    set(WJR_ENABLE_COVERAGE TRUE)
    set(CMAKE_DISABLE_FIND_PACKAGE_GTest TRUE)
    set(FETCHCONTENT_SOURCE_DIR_GOOGLETEST "${_SOURCE_DIR}")
endif()

if(MODE STREQUAL "local")
    # Exercise the conventional local source location without writing into the tree.
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/local")
    file(COPY "${_SOURCE_DIR}/" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/local/googletest")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/local/CMakeLists.txt" "wjr_load_gtest()\n")
    add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/local" local-build)
else()
    wjr_load_gtest()
    wjr_load_gtest()
endif()
if(NOT TARGET GTest::gtest_main OR NOT CMAKE_FIND_PACKAGE_TARGETS_GLOBAL)
    message(FATAL_ERROR "Provider target or parent discovery policy changed")
endif()
if(MODE STREQUAL "installed")
    if(NOT WJR_TEST_GTEST_PACKAGE_USED OR WJR_TEST_GTEST_SOURCE_USED)
        message(FATAL_ERROR "Compatible installed provider was not reused")
    endif()
    # All official exports must survive a repeated find in the caller's scope.
    find_package(GTest 1.17.0 EXACT REQUIRED CONFIG)
    foreach(_target GTest::gtest GTest::gtest_main GTest::gmock GTest::gmock_main)
        get_target_property(_GLOBAL ${_target} IMPORTED_GLOBAL)
        if(NOT _GLOBAL)
            message(FATAL_ERROR "Installed provider export is not globally visible")
        endif()
    endforeach()
    add_executable(provider_scope_probe
        "${CMAKE_CURRENT_LIST_DIR}/../../../cmake/gtest-provider/probe.cpp")
    target_link_libraries(provider_scope_probe PRIVATE GTest::gtest_main)
elseif(MODE MATCHES "^(preexisting|alias|bare|bare-alias)$")
    if(WJR_TEST_GTEST_PACKAGE_USED OR WJR_TEST_GTEST_SOURCE_USED)
        message(FATAL_ERROR "Parent target must bypass provider selection")
    endif()
    if(MODE STREQUAL "preexisting")
        get_target_property(_GLOBAL GTest::gtest_main IMPORTED_GLOBAL)
        if(_GLOBAL)
            message(FATAL_ERROR "Parent imported target scope changed")
        endif()
    endif()
elseif(NOT WJR_TEST_GTEST_SOURCE_USED)
    message(FATAL_ERROR "Source fallback was not used")
elseif(NOT MODE STREQUAL "broken" AND WJR_TEST_GTEST_PACKAGE_USED)
    message(FATAL_ERROR "Source selection unexpectedly loaded a package")
endif()
if(MODE MATCHES "^(installed|preexisting|alias|bare|bare-alias)$")
    if(NOT gtest_force_shared_crt STREQUAL "sentinel" OR
       NOT BUILD_GMOCK STREQUAL "sentinel" OR NOT BUILD_GTEST STREQUAL "sentinel")
        message(FATAL_ERROR "Reusing a provider changed parent source options")
    endif()
elseif(MODE STREQUAL "asan")
    get_target_property(_OPTIONS gtest COMPILE_OPTIONS)
    if(NOT "-fsanitize=address" IN_LIST _OPTIONS)
        message(FATAL_ERROR "Source provider lost ASan instrumentation")
    endif()
    if(WIN32 AND gtest_force_shared_crt)
        message(FATAL_ERROR "ASan source provider lost the static CRT policy")
    endif()
endif()
