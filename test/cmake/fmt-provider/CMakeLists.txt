cmake_minimum_required(VERSION 3.20)
project(wjr_fmt_provider_test LANGUAGES CXX)
include(CMakePackageConfigHelpers)

if(NOT MODE MATCHES "^(installed|preexisting|disabled|incompatible)$")
   message(FATAL_ERROR "Unknown fmt provider test mode: ${MODE}")
endif()

unset(WJR_TEST_FMT_PACKAGE_USED CACHE)
set(_PACKAGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/fake-fmt")
file(MAKE_DIRECTORY "${_PACKAGE_DIR}")
file(WRITE "${_PACKAGE_DIR}/fmt-config.cmake"
   "add_library(fmt::fmt INTERFACE IMPORTED GLOBAL)\n"
   "set(WJR_TEST_FMT_PACKAGE_USED TRUE CACHE INTERNAL \"test marker\")\n")
set(_VERSION 11.2.1)
if(MODE STREQUAL "incompatible")
   set(_VERSION 0.1.0)
endif()
write_basic_package_version_file("${_PACKAGE_DIR}/fmt-config-version.cmake"
   VERSION ${_VERSION} COMPATIBILITY SameMajorVersion)
set(fmt_DIR "${_PACKAGE_DIR}" CACHE PATH "Test fmt package" FORCE)

if(MODE STREQUAL "preexisting")
   add_library(fmt::fmt INTERFACE IMPORTED GLOBAL)
endif()
if(MODE MATCHES "^(preexisting|disabled)$")
   set(CMAKE_DISABLE_FIND_PACKAGE_fmt TRUE)
endif()

set(WJR_BUILD_TESTS OFF)
set(WJR_BUILD_BENCHMARKS OFF)
set(WJR_ENABLE_INSTALL OFF)
set(WJR_ENABLE_ASSEMBLY OFF)
set(FETCHCONTENT_TRY_FIND_PACKAGE_MODE NEVER)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../.." wjr)

if(NOT TARGET fmt::fmt)
   message(FATAL_ERROR "All supported provider modes must expose fmt::fmt")
endif()
get_target_property(_IMPORTED fmt::fmt IMPORTED)
if(MODE MATCHES "^(installed|preexisting)$")
   if(NOT _IMPORTED OR TARGET fmt)
      message(FATAL_ERROR "WJR rebuilt fmt instead of reusing the imported target")
   endif()
else()
   if(_IMPORTED OR NOT TARGET fmt)
      message(FATAL_ERROR "WJR did not fall back to its pinned bundled fmt")
   endif()
endif()
if(MODE STREQUAL "installed")
   if(NOT WJR_TEST_FMT_PACKAGE_USED)
      message(FATAL_ERROR "WJR did not load the compatible installed package")
   endif()
elseif(WJR_TEST_FMT_PACKAGE_USED)
   message(FATAL_ERROR "WJR loaded a disabled, unnecessary or incompatible package")
endif()
