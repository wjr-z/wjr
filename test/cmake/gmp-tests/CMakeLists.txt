cmake_minimum_required(VERSION 3.20)
project(wjr_gmp_test LANGUAGES NONE)

if(MODE MATCHES "^(switch|required-missing)$")
    set(_CONFIGURE_ARGS -G "${CMAKE_GENERATOR}")
    foreach(_ARG CMAKE_CXX_COMPILER CMAKE_MAKE_PROGRAM CMAKE_GENERATOR_PLATFORM
       CMAKE_GENERATOR_TOOLSET CMAKE_RC_COMPILER CMAKE_MT CMAKE_BUILD_TYPE)
        if(${_ARG})
            list(APPEND _CONFIGURE_ARGS "-D${_ARG}=${${_ARG}}")
        endif()
    endforeach()
    if(MODE STREQUAL "switch")
        set(_PHASES on off)
    else()
        set(_PHASES missing)
    endif()
    foreach(_PHASE IN LISTS _PHASES)
        execute_process(COMMAND "${CMAKE_COMMAND}"
            -S "${CMAKE_CURRENT_LIST_DIR}" -B "${CMAKE_CURRENT_BINARY_DIR}/child"
            ${_CONFIGURE_ARGS} "-DMODE=${_PHASE}"
            RESULT_VARIABLE _RESULT OUTPUT_VARIABLE _OUTPUT ERROR_VARIABLE _ERROR)
        if(MODE STREQUAL "required-missing")
            if(_RESULT EQUAL 0 OR NOT "${_OUTPUT}${_ERROR}" MATCHES
               "WJR_USE_GMP_TESTS=ON requires an available GMP")
                message(FATAL_ERROR "Missing GMP did not produce the expected diagnostic. ${_OUTPUT}${_ERROR}")
            endif()
        elseif(NOT _RESULT EQUAL 0)
            message(FATAL_ERROR "GMP ${_PHASE} configuration failed. ${_OUTPUT}${_ERROR}")
        endif()
    endforeach()
    return()
endif()

if(NOT MODE MATCHES "^(off|on|auto|pkgconfig|auto-missing|tool-missing|missing|legacy|legacy-off|explicit-off|package-preexisting-pkgconfig)$")
    message(FATAL_ERROR "Unknown GMP test mode ${MODE}")
endif()
enable_language(CXX)

unset(WJR_USE_GMP_TESTS)
unset(WJR_USE_GMP_TESTS CACHE)
unset(WJR_USE_GMP)
unset(WJR_USE_GMP CACHE)
if(MODE MATCHES "^(off|explicit-off)$")
    set(WJR_USE_GMP_TESTS OFF CACHE BOOL "Test option")
elseif(MODE MATCHES "^(on|missing)$")
    set(WJR_USE_GMP_TESTS ON CACHE BOOL "Test option")
endif()
if(MODE MATCHES "^(legacy|explicit-off)$")
    set(WJR_USE_GMP ON)
elseif(MODE STREQUAL "legacy-off")
    set(WJR_USE_GMP OFF)
endif()

set(_EXPECT_GMP ON)
if(MODE MATCHES "^(off|explicit-off|auto-missing|tool-missing)$")
    set(_EXPECT_GMP OFF)
endif()
set(GMP_LIBRARIES fixture_gmp CACHE STRING "Fixture GMP libraries")
set(GMP_INCLUDE_DIRS "${CMAKE_CURRENT_BINARY_DIR}/include" CACHE PATH "Fixture GMP headers")
set(_ORIGINAL_LIBRARIES "${GMP_LIBRARIES}")
set(_ORIGINAL_INCLUDES "${GMP_INCLUDE_DIRS}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules" "${CMAKE_CURRENT_BINARY_DIR}/include")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/gmp_fixture.h" "#define WJR_GMP_FIXTURE 1\n")
add_library(fixture_gmp INTERFACE)
if(MODE STREQUAL "package-preexisting-pkgconfig")
    add_library(PkgConfig::GMP INTERFACE IMPORTED GLOBAL)
endif()

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/modules/FindGMP.cmake" [=[
if(MODE MATCHES "^(off|explicit-off)$")
    message(FATAL_ERROR "Disabled GMP discovery was invoked")
endif()
set_property(GLOBAL APPEND PROPERTY WJR_TEST_GMP_CALLS GMP)
if(MODE MATCHES "^(pkgconfig|auto-missing|tool-missing|missing)$")
    set(GMP_FOUND FALSE)
else()
    set(GMP_FOUND TRUE)
endif()
]=])
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/modules/FindPkgConfig.cmake" [=[
if(MODE MATCHES "^(off|explicit-off)$")
    message(FATAL_ERROR "Disabled pkg-config discovery was invoked")
endif()
set_property(GLOBAL APPEND PROPERTY WJR_TEST_GMP_CALLS PkgConfig)
if(MODE STREQUAL "tool-missing")
    set(PkgConfig_FOUND FALSE)
    return()
endif()
set(PkgConfig_FOUND TRUE)
function(pkg_check_modules prefix)
    if(NOT "IMPORTED_TARGET" IN_LIST ARGN OR NOT "GLOBAL" IN_LIST ARGN)
        message(FATAL_ERROR "GMP pkg-config lookup must request a global imported target")
    endif()
    set_property(GLOBAL APPEND PROPERTY WJR_TEST_GMP_CALLS pkg_check_modules)
    if(MODE STREQUAL "pkgconfig")
        set(${prefix}_FOUND TRUE PARENT_SCOPE)
        add_library(PkgConfig::GMP INTERFACE IMPORTED GLOBAL)
        set_target_properties(PkgConfig::GMP PROPERTIES
            INTERFACE_INCLUDE_DIRECTORIES "${GMP_INCLUDE_DIRS}")
    else()
        set(${prefix}_FOUND FALSE PARENT_SCOPE)
    endif()
endfunction()
]=])
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_BINARY_DIR}/modules")
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/wjrGMP.cmake")

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/probe.cpp" [=[
#if WJR_GMP_EXPECTED
#ifndef WJR_USE_GMP
#error GMP definition missing
#endif
#include <gmp_fixture.h>
#elif defined(WJR_USE_GMP)
#error Disabled GMP definition leaked
#endif
int main() { return 0; }
]=])
foreach(_CONSUMER tests benchmarks)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${_CONSUMER}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${_CONSUMER}/CMakeLists.txt"
        "add_executable(${_CONSUMER} ../probe.cpp)\n"
        "wjr_link_gmp(${_CONSUMER})\n")
    add_subdirectory("${CMAKE_CURRENT_BINARY_DIR}/${_CONSUMER}" "${_CONSUMER}-build")
    target_compile_definitions(${_CONSUMER} PRIVATE "WJR_GMP_EXPECTED=$<BOOL:${_EXPECT_GMP}>")
    get_target_property(_LINKS ${_CONSUMER} LINK_LIBRARIES)
    get_target_property(_INCLUDES ${_CONSUMER} INCLUDE_DIRECTORIES)
    get_target_property(_DEFINITIONS ${_CONSUMER} COMPILE_DEFINITIONS)
    if(_EXPECT_GMP)
        if(NOT "wjr-test-gmp" IN_LIST _LINKS)
            message(FATAL_ERROR "${_CONSUMER} did not consume GMP")
        endif()
    elseif(_LINKS OR _INCLUDES OR "WJR_USE_GMP" IN_LIST _DEFINITIONS OR TARGET wjr-test-gmp)
        message(FATAL_ERROR "Disabled GMP leaked into ${_CONSUMER}")
    endif()
endforeach()

if(_EXPECT_GMP AND NOT WJR_USE_GMP_TESTS)
    message(FATAL_ERROR "Available GMP did not enable its requested/default option")
elseif(NOT _EXPECT_GMP AND WJR_USE_GMP_TESTS)
    message(FATAL_ERROR "Unavailable or disabled GMP was enabled")
endif()
if(NOT "${GMP_LIBRARIES}" STREQUAL _ORIGINAL_LIBRARIES OR
   NOT "${GMP_INCLUDE_DIRS}" STREQUAL _ORIGINAL_INCLUDES)
    message(FATAL_ERROR "GMP configuration changed preexisting dependency cache values")
endif()
get_property(_CALLS GLOBAL PROPERTY WJR_TEST_GMP_CALLS)
if(MODE MATCHES "^(off|explicit-off)$")
    set(_EXPECTED_CALLS "")
elseif(MODE STREQUAL "tool-missing")
    set(_EXPECTED_CALLS "GMP;PkgConfig")
elseif(MODE MATCHES "^(pkgconfig|auto-missing)$")
    set(_EXPECTED_CALLS "GMP;PkgConfig;pkg_check_modules")
else()
    set(_EXPECTED_CALLS "GMP")
endif()
if(NOT "${_CALLS}" STREQUAL _EXPECTED_CALLS)
    message(FATAL_ERROR "Unexpected dependency probes ${_CALLS}, expected ${_EXPECTED_CALLS}")
endif()
if(_EXPECT_GMP)
    get_target_property(_GMP_LINKS wjr-test-gmp INTERFACE_LINK_LIBRARIES)
    get_target_property(_GMP_DEFINITIONS wjr-test-gmp INTERFACE_COMPILE_DEFINITIONS)
    if(NOT "WJR_USE_GMP" IN_LIST _GMP_DEFINITIONS)
        message(FATAL_ERROR "Enabled GMP did not expose its compile definition")
    endif()
    if(MODE STREQUAL "pkgconfig")
        set(_EXPECTED_LINK PkgConfig::GMP)
    else()
        set(_EXPECTED_LINK fixture_gmp)
        get_target_property(_GMP_INCLUDES wjr-test-gmp INTERFACE_INCLUDE_DIRECTORIES)
        if(NOT GMP_INCLUDE_DIRS IN_LIST _GMP_INCLUDES)
            message(FATAL_ERROR "GMP package include directories were lost")
        endif()
    endif()
    if(NOT "${_GMP_LINKS}" STREQUAL _EXPECTED_LINK)
        message(FATAL_ERROR "GMP provider libraries were replaced or lost")
    endif()
endif()
if(MODE STREQUAL "pkgconfig")
    get_target_property(_GLOBAL PkgConfig::GMP IMPORTED_GLOBAL)
    if(NOT _GLOBAL)
        message(FATAL_ERROR "pkg-config target must be available to sibling consumers")
    endif()
endif()
