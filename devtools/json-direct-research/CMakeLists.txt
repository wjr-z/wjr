cmake_minimum_required(VERSION 3.20)
project(json_direct_research LANGUAGES CXX)
set(WJR_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(WJR_ENABLE_ASSEMBLY OFF CACHE BOOL "" FORCE)
add_subdirectory(${WJR_SOURCE} wjr EXCLUDE_FROM_ALL)
foreach(target json_direct_timing json_direct_allocations)
    add_executable(${target} driver.cpp)
    target_link_libraries(${target} PRIVATE wjr::json)
endforeach()
target_compile_definitions(json_direct_allocations PRIVATE COUNT_ALLOCATIONS=1)

find_package(GTest REQUIRED)
add_executable(json_direct_tests ${WJR_SOURCE}/test/json/direct_parser.cpp
    ${WJR_SOURCE}/test/json/pmr.cpp ${WJR_SOURCE}/test/json/registered_fields.cpp)
target_link_libraries(json_direct_tests PRIVATE wjr::json GTest::gtest_main)
add_executable(json_direct_sanitized ${WJR_SOURCE}/test/json/direct_parser.cpp)
target_link_libraries(json_direct_sanitized PRIVATE wjr::json GTest::gtest_main)
target_compile_options(json_direct_sanitized PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(json_direct_sanitized PRIVATE -fsanitize=address,undefined)

foreach(target json_direct_strong_timing json_direct_strong_allocations)
    add_executable(${target} strong-driver.cpp ${WJR_SOURCE}/examples/demo/src/simdjson.cpp)
    target_include_directories(${target} PRIVATE ${WJR_SOURCE}/examples/demo/src)
    target_link_libraries(${target} PRIVATE wjr::json)
endforeach()
target_compile_definitions(json_direct_strong_allocations PRIVATE COUNT_ALLOCATIONS=1)

add_executable(json_direct_hand hand-driver.cpp)
target_link_libraries(json_direct_hand PRIVATE wjr::json)
add_executable(json_direct_stages stages.cpp ${WJR_SOURCE}/examples/demo/src/simdjson.cpp)
target_include_directories(json_direct_stages PRIVATE ${WJR_SOURCE}/examples/demo/src)
target_link_libraries(json_direct_stages PRIVATE wjr::json)

foreach(target json_direct_matched_strong json_direct_matched_stages)
    if(target STREQUAL "json_direct_matched_strong")
        set(driver strong-driver.cpp)
    else()
        set(driver stages.cpp)
    endif()
    add_executable(${target} ${driver} ${WJR_SOURCE}/examples/demo/src/simdjson.cpp)
    target_include_directories(${target} PRIVATE ${WJR_SOURCE}/examples/demo/src)
    target_link_libraries(${target} PRIVATE wjr::json)
    target_compile_definitions(${target} PRIVATE SIMDJSON_IMPLEMENTATION_ICELAKE=0 SIMDJSON_IMPLEMENTATION_HASWELL=1)
endforeach()
