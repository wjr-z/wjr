cmake_minimum_required(VERSION 3.20)
project(json_immutable_research LANGUAGES CXX)
set(WJR_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(WJR_ENABLE_ASSEMBLY OFF CACHE BOOL "" FORCE)
add_subdirectory(${WJR_SOURCE} wjr EXCLUDE_FROM_ALL)
add_library(simdjson_reference STATIC ${WJR_SOURCE}/examples/demo/src/simdjson.cpp)
target_compile_features(simdjson_reference PUBLIC cxx_std_20)
target_compile_options(simdjson_reference PRIVATE -O3 -march=native)
target_include_directories(simdjson_reference PUBLIC ${WJR_SOURCE}/examples/demo/src)
foreach(target json_immutable_timing json_immutable_allocations)
    add_executable(${target} driver.cpp)
    target_compile_options(${target} PRIVATE -O3 -march=native)
    target_link_libraries(${target} PRIVATE wjr::json simdjson_reference)
endforeach()
target_compile_definitions(json_immutable_allocations PRIVATE COUNT_ALLOCATIONS=1)
find_package(GTest REQUIRED)
add_executable(json_immutable_tests ${WJR_SOURCE}/test/json/immutable_document.cpp
    ${WJR_SOURCE}/test/json/number_boundaries.cpp)
target_link_libraries(json_immutable_tests PRIVATE wjr::json GTest::gtest_main)
add_executable(json_immutable_sanitized ${WJR_SOURCE}/test/json/immutable_document.cpp
    ${WJR_SOURCE}/src/json/immutable_document.cpp ${WJR_SOURCE}/src/json/utf8.cpp)
target_link_libraries(json_immutable_sanitized PRIVATE wjr::json GTest::gtest_main)
target_compile_options(json_immutable_sanitized PRIVATE
    -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
target_link_options(json_immutable_sanitized PRIVATE -fsanitize=address,undefined)
