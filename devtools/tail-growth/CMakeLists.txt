cmake_minimum_required(VERSION 3.20)
project(tail_growth_research LANGUAGES CXX)
set(WJR_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(WJR_ENABLE_ASSEMBLY OFF CACHE BOOL "" FORCE)
set(WJR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(${WJR_SOURCE} wjr EXCLUDE_FROM_ALL)
add_executable(tail_timing timing.cpp observe.cpp)
target_link_libraries(tail_timing PRIVATE wjr::json)
if(GROWING_ALLOCATOR)
    target_compile_definitions(tail_timing PRIVATE GROWING_ALLOCATOR=1)
endif()
