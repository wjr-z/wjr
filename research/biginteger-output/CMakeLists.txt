cmake_minimum_required(VERSION 3.20)
project(scratch_probe LANGUAGES C CXX)
add_subdirectory("${WJR_SOURCE}" wjr EXCLUDE_FROM_ALL)
add_executable(scratch_probe probe.cpp)
target_link_libraries(scratch_probe PRIVATE wjr::biginteger)
target_compile_features(scratch_probe PRIVATE cxx_std_20)
add_executable(prefix_probe prefix_probe.cpp)
target_link_libraries(prefix_probe PRIVATE wjr::biginteger)
target_compile_features(prefix_probe PRIVATE cxx_std_20)

add_executable(timing_probe timing.cpp)
target_link_libraries(timing_probe PRIVATE wjr::biginteger)
target_compile_features(timing_probe PRIVATE cxx_std_20)

if(TARGET gtest_main)
  add_executable(output_tests "${WJR_SOURCE}/test/biginteger/to_chars_output.cpp")
  target_link_libraries(output_tests PRIVATE wjr::biginteger gtest_main)
  target_compile_features(output_tests PRIVATE cxx_std_20)
endif()
