cmake_minimum_required(VERSION 3.20)
project(compare128_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(BENCHMARK_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
add_subdirectory(${BENCHMARK_SOURCE} benchmark EXCLUDE_FROM_ALL)
foreach(side IN ITEMS old new)
 add_library(${side} OBJECT ${RESEARCH_SOURCE}/bridge.cpp)
 target_include_directories(${side} PRIVATE ${RESEARCH_SOURCE}/${side}_include ${WJR_SOURCE}/include)
 target_compile_definitions(${side} PRIVATE wjr=wjr_${side} WJR_VARIANT=${side} WJR_DEBUG_LEVEL=0)
endforeach()
add_executable(compare_probe ${RESEARCH_SOURCE}/benchmark.cpp $<TARGET_OBJECTS:old> $<TARGET_OBJECTS:new>)
target_link_libraries(compare_probe PRIVATE benchmark::benchmark)
