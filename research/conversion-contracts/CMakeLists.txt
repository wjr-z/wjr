cmake_minimum_required(VERSION 3.20)
project(conversion_contracts LANGUAGES C CXX)
add_subdirectory("${WJR_SOURCE}" wjr)
find_library(GMP_LIBRARY gmp REQUIRED)
foreach(name IN ITEMS contract_bench old_contract_bench)
  if(name STREQUAL "contract_bench")
    set(source "${WJR_SOURCE}/benchmarks/src/biginteger.cpp")
  else()
    set(source "${OLD_SOURCE}")
  endif()
  add_executable(${name} "${source}" contract.cpp main.cpp)
  target_include_directories(${name} PRIVATE "${WJR_SOURCE}/benchmarks/src")
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_compile_definitions(${name} PRIVATE WJR_USE_GMP)
  target_link_libraries(${name} PRIVATE wjr::biginteger benchmark::benchmark "${GMP_LIBRARY}")
  target_link_options(${name} PRIVATE -Wl,--wrap=__gmpn_get_str -Wl,--wrap=__gmpn_set_str)
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/paths.txt" CONTENT "$<TARGET_FILE:benchmarks>\n$<TARGET_FILE:contract_bench>\n$<TARGET_FILE:old_contract_bench>\n")
