cmake_minimum_required(VERSION 3.20)
project(wjr_expression_experiment LANGUAGES CXX)
include(CTest)

if(NOT DEFINED CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 20)
endif()
if(CMAKE_CXX_STANDARD LESS 20)
    message(FATAL_ERROR "This experiment requires C++20 or later.")
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WJR_EXPR_XSIMD "Probe the pinned optional xsimd evaluator." OFF)
option(WJR_EXPR_BENCHMARK "Build the Google Benchmark comparison." OFF)

add_library(expression_codegen OBJECT validation.cpp)
add_executable(expression_example main.cpp)
set(_WJR_EXPRESSION_TARGETS expression_codegen expression_example)
if(BUILD_TESTING)
    add_executable(expression_test validation.cpp)
    target_compile_definitions(expression_test PRIVATE WJR_EXPR_TEST)
    list(APPEND _WJR_EXPRESSION_TARGETS expression_test)
    add_test(NAME expression_value_closure COMMAND expression_test)
    add_test(NAME expression_syntax_example COMMAND expression_example)
endif()

if(WJR_EXPR_BENCHMARK)
    if(NOT TARGET benchmark::benchmark)
        find_package(benchmark 1.9 CONFIG REQUIRED)
    endif()
    add_executable(expression_benchmark validation.cpp)
    target_compile_definitions(expression_benchmark PRIVATE WJR_EXPR_BENCHMARK)
    target_link_libraries(expression_benchmark PRIVATE benchmark::benchmark)
    list(APPEND _WJR_EXPRESSION_TARGETS expression_benchmark)
endif()

if(WJR_EXPR_XSIMD AND NOT TARGET xsimd)
    find_package(xsimd 14.3.0 EXACT CONFIG REQUIRED)
endif()
foreach(_WJR_TARGET IN LISTS _WJR_EXPRESSION_TARGETS)
    target_include_directories(${_WJR_TARGET} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../include")
    target_compile_definitions(${_WJR_TARGET} PRIVATE WJR_DEBUG_LEVEL=0)
    if(MSVC)
        target_compile_options(${_WJR_TARGET} PRIVATE /EHsc /Zc:preprocessor /utf-8)
    endif()
    if(WJR_EXPR_XSIMD)
        target_compile_definitions(${_WJR_TARGET} PRIVATE WJR_EXPR_XSIMD)
        target_link_libraries(${_WJR_TARGET} PRIVATE xsimd)
    endif()
endforeach()
