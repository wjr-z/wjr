cmake_minimum_required(VERSION 3.22)
project(wjr_np_examples LANGUAGES CXX)

find_package(wjr CONFIG REQUIRED COMPONENTS np)

option(WJR_NP_EXAMPLES_XSIMD "Build the explicit SIMD expression example" OFF)
if(WJR_NP_EXAMPLES_XSIMD)
    find_package(wjr CONFIG REQUIRED COMPONENTS simd_xsimd)
endif()
option(WJR_NP_EXAMPLES_CBLAS "Build the explicit CBLAS example" OFF)
if(WJR_NP_EXAMPLES_CBLAS)
    find_package(wjr CONFIG REQUIRED COMPONENTS np_cblas)
endif()

include(CTest)
file(GLOB NP_EXAMPLE_SRCS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
foreach(source IN LISTS NP_EXAMPLE_SRCS)
    get_filename_component(name "${source}" NAME_WE)
    if(name STREQUAL "expression_simd" AND NOT WJR_NP_EXAMPLES_XSIMD)
        continue()
    endif()
    if(name STREQUAL "cblas" AND NOT WJR_NP_EXAMPLES_CBLAS)
        continue()
    endif()
    set(target "wjr_np_${name}")
    add_executable(${target} "${source}")
    target_link_libraries(${target} PRIVATE wjr::np)
    if(name STREQUAL "expression_simd")
        target_link_libraries(${target} PRIVATE wjr::simd_xsimd)
    endif()
    if(name STREQUAL "cblas")
        target_link_libraries(${target} PRIVATE wjr::np_cblas)
    endif()
    if(BUILD_TESTING)
        add_test(NAME ${target} COMMAND ${target})
    endif()
endforeach()
